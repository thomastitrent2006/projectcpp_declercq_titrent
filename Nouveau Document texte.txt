cmake_minimum_required(VERSION 3.15)
project(hello VERSION 0.1 LANGUAGES CXX)

# indique où chercher les headers installés pour l'IDE (optionnel)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# créer une bibliothèque (ou on peut directement créer un executable)
add_library(hello_lib src/hello.cpp)
target_include_directories(hello_lib PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)

add_executable(hello_app src/main.cpp)
target_link_libraries(hello_app PRIVATE hello_lib)

install(TARGETS hello_app RUNTIME DESTINATION bin)
install(TARGETS hello_lib ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)

enable_testing()
add_executable(test_hello tests/test_main.cpp)
target_link_libraries(test_hello PRIVATE hello_lib)
add_test(NAME HelloTest COMMAND test_hello)

cmake_minimum_required(VERSION 3.10)
project(TestCMake C)

add_executable(test_program main.c)
